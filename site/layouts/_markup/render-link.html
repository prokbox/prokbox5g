{{- /* Links between Markdown files (../guide/music.md#section) become links to the generated pages, so the same source reads well on GitHub. */ -}} {{- /* An empty link to #item-... is an entry's anchor on What's coming (scripts/roadmap.py), which the home page's lanes link to: it becomes the anchor itself */ -}} {{- if and (eq (string .Text) "") (or (hasPrefix .Destination "#item-") (hasPrefix .Destination "#notplanned-")) -}} {{- else -}} {{- $dest := .Destination -}} {{- $u := urls.Parse $dest -}} {{- if and (not $u.IsAbs) (strings.HasSuffix $u.Path ".md") -}} {{- $target := path.Join .Page.File.Dir $u.Path | path.Clean -}} {{- with site.GetPage $target -}} {{- $dest = .RelPermalink -}} {{- with $u.Fragment }}{{ $dest = printf "%s#%s" $dest . }}{{ end -}} {{- else -}} {{- errorf "%s: broken link to %s" $.Page.File.Path $.Destination -}} {{- end -}} {{- end -}} {{ .Text }} {{- end -}} {{- /**/ -}}
